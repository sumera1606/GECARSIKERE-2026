rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Students: anyone authenticated can read; only admin can write
    match /students/{studentId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == studentId;
      allow update, delete: if request.auth != null && (
        request.auth.token.email == "admin@gecarsikere.edu"
      );
    }

    // Pending students: anyone can create (registration); only admin can read/delete
    match /pendingStudents/{studentId} {
      allow read: if request.auth != null && (
        request.auth.uid == studentId ||
        request.auth.token.email == "admin@gecarsikere.edu"
      );
      allow create: if request.auth != null && request.auth.uid == studentId;
      allow delete: if request.auth != null && (
        request.auth.token.email == "admin@gecarsikere.edu"
      );
    }

    // Faculty: anyone authenticated can read; self can create own doc
    match /faculty/{facultyId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == facultyId;
      allow update: if request.auth != null && (
        request.auth.uid == facultyId ||
        request.auth.token.email == "admin@gecarsikere.edu"
      );
      allow delete: if request.auth != null && (
        request.auth.token.email == "admin@gecarsikere.edu"
      );
    }

    // Announcements: anyone authenticated can read; only admin can write
    match /announcements/{announcementId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && (
        request.auth.token.email == "admin@gecarsikere.edu"
      );
    }

    // Assignments: anyone authenticated can read; faculty can create
    match /assignments/{assignmentId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow delete, update: if request.auth != null && (
        request.auth.token.email == "admin@gecarsikere.edu" ||
        resource.data.postedBy == request.auth.uid
      );
    }

    // Attendance: anyone authenticated can read; faculty can write
    match /attendance/{attendanceId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    // Messages: participants can read/write their own conversations
    match /messages/{messageId} {
      allow read: if request.auth != null && (
        resource.data.studentId == request.auth.uid ||
        resource.data.facultyId == request.auth.uid ||
        request.auth.token.email == "admin@gecarsikere.edu"
      );
      allow create: if request.auth != null && (
        request.auth.uid == request.resource.data.studentId ||
        request.auth.uid == request.resource.data.facultyId
      );
      allow delete: if request.auth != null && (
        request.auth.token.email == "admin@gecarsikere.edu"
      );
    }

    // Media: anyone authenticated can read; only admin can write
    match /media/{mediaId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && (
        request.auth.token.email == "admin@gecarsikere.edu"
      );
    }
  }
}
